# Pipedrive MCP Configuration
# Set these environment variables in your .env file

# Get these from your Pipedrive account settings
PIPEDRIVE_DOMAIN=your-company-domain  # e.g., "mycompany" for mycompany.pipedrive.com
PIPEDRIVE_API_KEY=your_api_key_here    # Found in Settings > Personal preferences > API

# Example .env file:
# PIPEDRIVE_DOMAIN=mycompany
# PIPEDRIVE_API_KEY=1234567890abcdef1234567890abcdef12345678

# Steps to get your credentials:
# 1. Log into your Pipedrive account
# 2. Go to Settings (gear icon) > Personal preferences > API
# 3. Copy your API token
# 4. Your domain is the part before .pipedrive.com in your account URL
# 5. Create .env file with the variables above
